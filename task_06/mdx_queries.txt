1st Query
WITH
    MEMBER [Measures].[Portugal GDP per capita] AS
        ([Measures].[GDP per capita], [Country].[Country].[Portugal])
    MEMBER [Measures].[Spain GDP per capita] AS
        ([Measures].[GDP per capita], [Country].[Country].[Spain])
    MEMBER [Measures].[Portugal to Spain ratio] AS
        [Measures].[Portugal GDP per capita] / [Measures].[Spain GDP per capita],
        FORMAT_STRING = '#0.00%'
SELECT
    {[Measures].[Portugal GDP per capita], [Measures].[Spain GDP per capita], [Measures].[Portugal to Spain ratio]} ON COLUMNS,
    {[Time].[Decade].[1960], [Time].[Decade].[1970], [Time].[Decade].[1980],
    [Time].[Decade].[1990], [Time].[Decade].[2000], [Time].[Decade].[2010],
    [Time].[Decade].[2020]} ON ROWS
FROM [Economy]

2nd Query
SELECT {[Time].[Year].[1995], [Time].[Year].[2020]} ON COLUMNS,
    GENERATE([Sector].[Broad].Members,
        DESCENDANTS([Sector].CurrentMember, [Sector].[Sector], SELF_AND_BEFORE)) ON ROWS
FROM [Sector]
WHERE ([Country].[Country].[Portugal], [Measures].[Value Added PCT])
