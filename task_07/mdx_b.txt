WITH MEMBER [Measures].[GDP per capita relative to Portugal] AS
    [Measures].[GDP per capita] /
    ([Measures].[GDP per capita], [Country].[Country].[Portugal]),
    FORMAT_STRING = '#0.0%'
SELECT [Time].[Decade].Members ON COLUMNS,
    [Country].[Country Group].[Western Europe].Children ON ROWS
FROM [Economy]
WHERE [Measures].[GDP per capita relative to Portugal]
