query 1 Economy
WITH
    SET [Countries] AS
        {[Country].[Country].[Portugal],
         [Country].[Country Group].[Central/Eastern Europe].Children,
         [Country].[Country Group].[Post-Soviet].Children}
    MEMBER [Measures].[GDP per capita relative to Portugal] AS
        [Measures].[GDP per capita] /
        ([Measures].[GDP per capita], [Country].[Country].[Portugal]),
        FORMAT_STRING = '#0.0%'
    MEMBER [Measures].[GDP per capita as percentage compared to 1990] AS
        [Measures].[GDP per capita] /
        ([Measures].[GDP per capita], [Time].[Decade].[1990]),
        FORMAT_STRING = '#0%'
    MEMBER [Measures].[Trade Openness] AS
        IIF(IsEmpty([Measures].[Exports]) OR IsEmpty([Measures].[Imports]), NULL,
            [Measures].[Exports] + [Measures].[Imports]),
        FORMAT_STRING = '#0.0'
SELECT
    {[Time].[Decade].[1990] : [Time].[Decade].[2020]} ON COLUMNS,
    NON EMPTY CrossJoin([Countries],
        {[Measures].[GDP per capita], [Measures].[GDP per capita relative to Portugal],
         [Measures].[GDP per capita as percentage compared to 1990], 
         [Measures].[Trade Openness]}) ON ROWS
FROM [Economy]

query 2 Society
WITH
    SET [Countries] AS
        {[Country].[Country].[Portugal],
            [Country].[Country Group].[Central/Eastern Europe].Children,
            [Country].[Country Group].[Post-Soviet].Children}
    MEMBER [Measures].[GNI per capita relative to Portugal] AS
        [Measures].[GNI per capita] /
        ([Measures].[GNI per capita], [Country].[Country].[Portugal]),
        FORMAT_STRING = '#0.0%'
    MEMBER [Measures].[Life expectancy gap vs Portugal] AS
        [Measures].[Life expectancy] -
        ([Measures].[Life expectancy], [Country].[Country].[Portugal]),
        FORMAT_STRING = '+#0.0;-#0.0'
SELECT
    {[Time].[Decade].[1990] : [Time].[Decade].[2020]} ON COLUMNS,
    NON EMPTY CrossJoin([Countries],
        {[Measures].[GNI per capita], [Measures].[GNI per capita relative to Portugal],
            [Measures].[Life expectancy], [Measures].[Life expectancy gap vs Portugal],
            [Measures].[Urban population]}) ON ROWS
FROM [Society]
