WITH SET [Comparison Countries] AS
    {[Country].[Country].[Portugal],
    [Country].[Country Group].[Central/Eastern Europe].Children,
    [Country].[Country Group].[Post-Soviet].Children}
SELECT TAIL([Time].[Decade].Members, 4) ON COLUMNS,
    [Comparison Countries] ON ROWS
FROM [Economy]
WHERE [Measures].[GDP per capita]
