WITH
    MEMBER [Measures].[PT GDP per capita] AS
        ([Measures].[GDP per capita], [Country].[Country].[Portugal])
    MEMBER [Measures].[Western Europe GDP per capita] AS
        ([Measures].[GDP per capita], [Country].[Country Group].[Western Europe])
    MEMBER [Measures].[PT to Western Europe ratio] AS
        [Measures].[PT GDP per capita] / [Measures].[Western Europe GDP per capita],
        FORMAT_STRING = '#0.00%'
SELECT
    {[Measures].[PT GDP per capita], [Measures].[Western Europe GDP per capita], [Measures].[PT to Western Europe ratio]} ON COLUMNS,
    [Time].[Decade].Members ON ROWS
FROM [Economy]
