query 1 Economy
WITH
    SET [WE] AS [Country].[Country Group].[Western Europe].Children
    SET [Countries] AS {[Country].[Country].[Portugal], [WE]}
    MEMBER [Measures].[Portugal GDP per capita compared to the country] AS
        IIF(IsEmpty([Measures].[GDP per capita]), NULL,
            IIF([Country].CurrentMember IS [Country].[Country].[Portugal],
                [Measures].[GDP per capita] / Avg([WE], [Measures].[GDP per capita]),
                ([Measures].[GDP per capita], [Country].[Country].[Portugal]) /
                [Measures].[GDP per capita])),
        FORMAT_STRING = '#0.0%'
    MEMBER [Measures].[Percentual change] AS
        IIF(IsEmpty(([Measures].[Portugal GDP per capita compared to the country], [Time].CurrentMember.PrevMember)), NULL,
            ([Measures].[Portugal GDP per capita compared to the country] -
                ([Measures].[Portugal GDP per capita compared to the country], [Time].CurrentMember.PrevMember))),
        FORMAT_STRING = '+#0.0%;-#0.0%'
SELECT
    {[Time].[Decade].[1900] : [Time].[Decade].[2020]} ON COLUMNS,
    NON EMPTY CrossJoin([Countries],
        {[Measures].[GDP per capita], [Measures].[Portugal GDP per capita compared to the country],
            [Measures].[Percentual change]}) ON ROWS
FROM [Economy]

Query 2 Society
WITH
    SET [WE] AS {[Country].[Country].[Portugal],
                 [Country].[Country Group].[Western Europe].Children}
    MEMBER [Measures].[Portugal GNI per capita compared to the country] AS
        ([Measures].[GNI per capita], [Country].[Country].[Portugal]) /
        [Measures].[GNI per capita],
        FORMAT_STRING = '#0.0%'
    MEMBER [Measures].[Percentual change] AS
        IIF(IsEmpty(([Measures].[Portugal GNI per capita compared to the country], [Time].CurrentMember.PrevMember)), NULL,
            ([Measures].[Portugal GNI per capita compared to the country] -
                ([Measures].[Portugal GNI per capita compared to the country], [Time].CurrentMember.PrevMember))),
        FORMAT_STRING = '+#0.0%;-#0.0%'
    MEMBER [Measures].[Portugal life expectancy gap (in years)] AS
        ([Measures].[Life expectancy], [Country].[Country].[Portugal]) -
        [Measures].[Life expectancy],
        FORMAT_STRING = '+#0.0;-#0.0'
SELECT
    {[Time].[Decade].[1960] : [Time].[Decade].[2020]} ON COLUMNS,
    NON EMPTY CrossJoin([WE],
        {[Measures].[Portugal GNI per capita compared to the country], [Measures].[Percentual change],
            [Measures].[Portugal life expectancy gap (in years)]}) ON ROWS
FROM [Society]
